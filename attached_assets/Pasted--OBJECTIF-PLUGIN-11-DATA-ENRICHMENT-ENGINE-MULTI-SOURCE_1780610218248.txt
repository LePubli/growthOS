# OBJECTIF : PLUGIN 11 - DATA ENRICHMENT ENGINE MULTI-SOURCES
Créer un moteur d'enrichissement de données B2B avec 23 sources (APIs + scraping), configurable via les paramètres (APIs payantes), et interconnecté avec TOUS les plugins GrowthOS.

# PHASE 1 : BASE DE DONNÉES

## 1.1 Migration pour les sources et configurations
Créer : `artifacts/api-server/db/migrations/011_enrichment_engine.sql`

```sql
-- Table des configurations d'APIs (payantes et gratuites)
CREATE TABLE IF NOT EXISTS enrichment_api_configs (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  source_id TEXT NOT NULL,
  api_key TEXT,
  api_secret TEXT,
  endpoint_url TEXT,
  rate_limit INTEGER DEFAULT 60,
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(source_id)
);

-- Table des données enrichies
CREATE TABLE IF NOT EXISTS enrichment_data (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  prospect_id UUID REFERENCES prospects(id) ON DELETE CASCADE,
  source_id TEXT NOT NULL,
  data_type TEXT NOT NULL, -- 'legal', 'financial', 'digital', 'social', 'news', 'jobs'
  data JSONB NOT NULL,
  confidence_score FLOAT DEFAULT 0.0,
  fetched_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(prospect_id, source_id, data_type)
);

-- Table des signaux détectés
CREATE TABLE IF NOT EXISTS enrichment_signals (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  prospect_id UUID REFERENCES prospects(id) ON DELETE CASCADE,
  signal_type TEXT NOT NULL, -- 'hiring', 'funding', 'news', 'tech_change', 'leadership_change'
  title TEXT NOT NULL,
  description TEXT,
  source TEXT NOT NULL,
  impact_score INTEGER DEFAULT 50,
  detected_at TIMESTAMP DEFAULT NOW(),
  processed BOOLEAN DEFAULT FALSE
);

-- Index
CREATE INDEX idx_enrichment_prospect ON enrichment_data(prospect_id);
CREATE INDEX idx_enrichment_source ON enrichment_data(source_id);
CREATE INDEX idx_signals_prospect ON enrichment_signals(prospect_id);
CREATE INDEX idx_signals_type ON enrichment_signals(signal_type);
